⚠️ Convenciones importantes

  • IDs únicos: Cada funcionalidad tiene un ID (ej: EQ-001, FIC-001) que debe usarse como referencia

  • Autenticación: Todas las peticiones requieren Authorization: Bearer {token}
  • Empresa context: Todas las peticiones incluyen header X-Company-Id: {companyId}
  • Formato fechas: ISO 8601 (YYYY-MM-DDTHH:mm:ssZ)
  • Formato moneda: Decimal (2 decimales, ej: 1500.50)

GET /employees

9. ESQUEMA DE BASE DE DATOS PROPUESTO

📋 Tablas principales del módulo Equipo

A continuación se detalla el esquema de base de datos sugerido para implementar todas las funcionalidades del módulo.

Tabla: employees

CREATE TABLE employees (
  id VARCHAR(50) PRIMARY KEY,
  company_id VARCHAR(50) NOT NULL,
  full_name VARCHAR(255) NOT NULL,
  personal_email VARCHAR(255) UNIQUE,
  professional_email VARCHAR(255) UNIQUE,
  dni VARCHAR(50) NOT NULL,
  date_of_birth DATE,
  place_of_birth VARCHAR(255),
  nationality VARCHAR(2),
  address_street VARCHAR(255),
  address_city VARCHAR(100),
  address_postal_code VARCHAR(20),
  address_country VARCHAR(2),
  phone VARCHAR(50),
  iban VARCHAR(50),
  emergency_contact_name VARCHAR(255),
  emergency_contact_phone VARCHAR(50),
  
  contract_type VARCHAR(50),
  working_day VARCHAR(50),
  weekly_hours DECIMAL(5,2),
  gross_salary DECIMAL(10,2),
  work_center_id VARCHAR(50),
  start_date DATE,
  collective_agreement VARCHAR(100),
  
  social_security_number VARCHAR(50),
  sepe_contract_code VARCHAR(20),
  social_security_date DATE,
  cotization_group VARCHAR(10),
  cotization_type VARCHAR(10),
  mutua VARCHAR(100),
  company_ccc VARCHAR(20),
  irpf_percentage DECIMAL(5,2),
  
  role_id VARCHAR(50) NOT NULL,
  status VARCHAR(20) DEFAULT 'active',
  
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  created_by VARCHAR(50),
  
  FOREIGN KEY (company_id) REFERENCES companies(id),
  FOREIGN KEY (role_id) REFERENCES roles(id),
  FOREIGN KEY (work_center_id) REFERENCES work_centers(id),
  
  INDEX idx_company_status (company_id, status),
  INDEX idx_email (personal_email, professional_email)
);

Tabla: clock_ins

CREATE TABLE clock_ins (
  id VARCHAR(50) PRIMARY KEY,
  employee_id VARCHAR(50) NOT NULL,
  company_id VARCHAR(50) NOT NULL,
  date DATE NOT NULL,
  clock_in_time TIMESTAMP,
  clock_out_time TIMESTAMP,
  total_hours DECIMAL(5,2),
  status VARCHAR(20),
  punctuality VARCHAR(20),
  device VARCHAR(20),
  geolocation_lat DECIMAL(10,8),
  geolocation_lng DECIMAL(11,8),
  geolocation_accuracy DECIMAL(6,2),
  manually_adjusted BOOLEAN DEFAULT FALSE,
  adjusted_by VARCHAR(50),
  adjustment_reason TEXT,
  
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  FOREIGN KEY (employee_id) REFERENCES employees(id),
  FOREIGN KEY (company_id) REFERENCES companies(id),
  
  INDEX idx_employee_date (employee_id, date),
  INDEX idx_company_date (company_id, date)
);

Tabla: vacation_requests

CREATE TABLE vacation_requests (
  id VARCHAR(50) PRIMARY KEY,
  employee_id VARCHAR(50) NOT NULL,
  company_id VARCHAR(50) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  working_days INT NOT NULL,
  comments TEXT,
  half_day BOOLEAN DEFAULT FALSE,
  status VARCHAR(20) DEFAULT 'pending',
  requested_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  approved_by VARCHAR(50),
  approved_at TIMESTAMP,
  rejection_reason TEXT,
  type VARCHAR(20) DEFAULT 'employee_request',
  
  FOREIGN KEY (employee_id) REFERENCES employees(id),
  FOREIGN KEY (company_id) REFERENCES companies(id),
  FOREIGN KEY (approved_by) REFERENCES employees(id),
  
  INDEX idx_employee_status (employee_id, status),
  INDEX idx_company_dates (company_id, start_date, end_date)
);

Tabla: expenses

CREATE TABLE expenses (
  id VARCHAR(50) PRIMARY KEY,
  employee_id VARCHAR(50) NOT NULL,
  company_id VARCHAR(50) NOT NULL,
  date DATE NOT NULL,
  category VARCHAR(50) NOT NULL,
  amount DECIMAL(10,2) NOT NULL,
  concept VARCHAR(255) NOT NULL,
  justification TEXT,
  receipt_url VARCHAR(500),
  status VARCHAR(20) DEFAULT 'pending',
  submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  reviewed_by VARCHAR(50),
  reviewed_at TIMESTAMP,
  rejection_reason TEXT,
  
  FOREIGN KEY (employee_id) REFERENCES employees(id),
  FOREIGN KEY (company_id) REFERENCES companies(id),
  
  INDEX idx_employee_status (employee_id, status),
  INDEX idx_company_date (company_id, date)
);

Tabla: consumptions

CREATE TABLE consumptions (
  id VARCHAR(50) PRIMARY KEY,
  employee_id VARCHAR(50) NOT NULL,
  company_id VARCHAR(50) NOT NULL,
  date DATE NOT NULL,
  category VARCHAR(50) NOT NULL,
  quantity DECIMAL(10,2) NOT NULL,
  description VARCHAR(255),
  status VARCHAR(20) DEFAULT 'pending',
  submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  reviewed_by VARCHAR(50),

  FOREIGN KEY (employee_id) REFERENCES employees(id),
  FOREIGN KEY (company_id) REFERENCES companies(id),
  
  INDEX idx_employee_date (employee_id, date),
  INDEX idx_company_status (company_id, status)
);

Tabla: documents

CREATE TABLE documents (
  id VARCHAR(50) PRIMARY KEY,
  employee_id VARCHAR(50) NOT NULL,
  company_id VARCHAR(50) NOT NULL,
  document_type VARCHAR(50) NOT NULL,
  file_name VARCHAR(255) NOT NULL,
  file_url VARCHAR(500) NOT NULL,
  file_size INT,
  status VARCHAR(20) DEFAULT 'pending_review',
  uploaded_by VARCHAR(50) NOT NULL,
  uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (employee_id) REFERENCES employees(id),
  FOREIGN KEY (company_id) REFERENCES companies(id),
  
  INDEX idx_employee_type (employee_id, document_type)

Tabla: schedules

CREATE TABLE schedules (
  id VARCHAR(50) PRIMARY KEY,
  employee_id VARCHAR(50) NOT NULL,
  company_id VARCHAR(50) NOT NULL,
  date DATE NOT NULL,
  shift_id VARCHAR(50),

// Middleware de validación de permisos
function checkPermission(requiredPermission, requiredLevel) {
  return async (req, res, next) => {
    const userId = req.user.id;
    const user = await User.findById(userId).populate('role');
    
    // Verificar que el rol tenga el permiso
    const hasPermission = user.role.permissions.includes(requiredPermission);
    
    if (!hasPermission) {
      return res.status(403).json({
        success: false,
        error: {
          code: 'FORBIDDEN',
          message: 'No tienes permisos para realizar esta acción'
        }
      });
    }
    
    next();
  };
}

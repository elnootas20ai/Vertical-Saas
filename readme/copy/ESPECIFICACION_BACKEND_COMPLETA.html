📋 Convenciones API

  • Base URL: https://api.udaredge.com/v1
  • Autenticación: Header Authorization: Bearer {token}
  • Context empresa: Header X-Company-Id: {companyId}
  • Fechas: ISO 8601 (YYYY-MM-DDTHH:mm:ssZ)
  • Moneda: Decimal 2 decimales (ej: 1500.50)

✅ Estructura de respuesta estándar

Error:

EQ-008 | Crear nuevo empleado
POST /employees

Request Body:

{ "personalData": { "fullName": "María González López", "personalEmail": "maria@gmail.com", "professionalEmail": "maria.gonzalez@empresa.com", "dni": "12345678Z", "dateOfBirth": "1990-05-15", "address": { "street": "Calle Mayor 123", "city": "Madrid", "postalCode": "28001", "country": "ES" }, "phone": "+34612345678", "iban": "ES9121000418450200051332", "emergencyContact": { "name": "Pedro González", "phone": "+34698765432" } }, "laborData": { "contractType": "indefinido", "workingDay": "completa", "weeklyHours": 40, "grossSalary": 2500.00, "workCenterId": "wc_789ghi", "startDate": "2026-03-01", "collectiveAgreement": "comercio_madrid" }, "administrativeData": { "socialSecurityNumber": "281234567890", "sepeContractCode": "100", "socialSecurityDate": "2026-03-01", "cotizationGroup": "5", "mutua": "Mutua MAZ", "companyCCC": "28123456789", "irpfPercentage": 15.5 }, "roleId": "role_456def" }

Validaciones:

  • Email único (personal y profesional)
  • DNI formato válido según país
  • Salario >= salario mínimo del país
  • Horas semanales <= 40h (España)
  • IBAN formato válido
  • roleId y workCenterId deben existir

Acciones automáticas:

  • Asignar permisos del rol
  • Si FLAG onboarding: enviar invitación + crear checklist
  • Si FLAG audit: registrar en historial
  • Si FLAG gestoria: notificar alta

9. ESQUEMA DE BASE DE DATOS

Tabla: employees

CREATE TABLE employees ( id VARCHAR(50) PRIMARY KEY, company_id VARCHAR(50) NOT NULL, full_name VARCHAR(255) NOT NULL, personal_email VARCHAR(255) UNIQUE, professional_email VARCHAR(255) UNIQUE, dni VARCHAR(50) NOT NULL, date_of_birth DATE, address_street VARCHAR(255), address_city VARCHAR(100), address_postal_code VARCHAR(20), phone VARCHAR(50), iban VARCHAR(50), emergency_contact_name VARCHAR(255), emergency_contact_phone VARCHAR(50), contract_type VARCHAR(50), weekly_hours DECIMAL(5,2), gross_salary DECIMAL(10,2), work_center_id VARCHAR(50), start_date DATE, collective_agreement VARCHAR(100), social_security_number VARCHAR(50), irpf_percentage DECIMAL(5,2), role_id VARCHAR(50) NOT NULL, status VARCHAR(20) DEFAULT 'active', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, FOREIGN KEY (company_id) REFERENCES companies(id), FOREIGN KEY (role_id) REFERENCES roles(id), INDEX idx_company_status (company_id, status) );

Tabla: clock_ins

CREATE TABLE clock_ins ( id VARCHAR(50) PRIMARY KEY, employee_id VARCHAR(50) NOT NULL, company_id VARCHAR(50) NOT NULL, date DATE NOT NULL, clock_in_time TIMESTAMP, clock_out_time TIMESTAMP, total_hours DECIMAL(5,2), status VARCHAR(20), punctuality VARCHAR(20), device VARCHAR(20), geolocation_lat DECIMAL(10,8), geolocation_lng DECIMAL(11,8), manually_adjusted BOOLEAN DEFAULT FALSE, adjusted_by VARCHAR(50), adjustment_reason TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (employee_id) REFERENCES employees(id), INDEX idx_employee_date (employee_id, date) );

Tabla: vacation_requests

CREATE TABLE vacation_requests ( id VARCHAR(50) PRIMARY KEY, employee_id VARCHAR(50) NOT NULL, start_date DATE NOT NULL, end_date DATE NOT NULL, working_days INT NOT NULL, comments TEXT, status VARCHAR(20) DEFAULT 'pending', requested_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, approved_by VARCHAR(50), approved_at TIMESTAMP, rejection_reason TEXT, FOREIGN KEY (employee_id) REFERENCES employees(id), INDEX idx_employee_status (employee_id, status) );

Tabla: expenses

CREATE TABLE expenses ( id VARCHAR(50) PRIMARY KEY, employee_id VARCHAR(50) NOT NULL, date DATE NOT NULL, category VARCHAR(50) NOT NULL, amount DECIMAL(10,2) NOT NULL, concept VARCHAR(255) NOT NULL, receipt_url VARCHAR(500), status VARCHAR(20) DEFAULT 'pending', submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, reviewed_by VARCHAR(50), FOREIGN KEY (employee_id) REFERENCES employees(id), INDEX idx_employee_status (employee_id, status) );

Tabla: documents

CREATE TABLE documents ( id VARCHAR(50) PRIMARY KEY, employee_id VARCHAR(50) NOT NULL, document_type VARCHAR(50) NOT NULL, file_name VARCHAR(255) NOT NULL, file_url VARCHAR(500) NOT NULL, file_size INT, status VARCHAR(20) DEFAULT 'pending_review', uploaded_by VARCHAR(50) NOT NULL, uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (employee_id) REFERENCES employees(id), INDEX idx_employee_type (employee_id, document_type) );

Tabla: schedules

CREATE TABLE schedules ( id VARCHAR(50) PRIMARY KEY, employee_id VARCHAR(50) NOT NULL, date DATE NOT NULL, shift_id VARCHAR(50), is_day_off BOOLEAN DEFAULT FALSE, FOREIGN KEY (employee_id) REFERENCES employees(id), FOREIGN KEY (shift_id) REFERENCES shifts(id), UNIQUE KEY unique_employee_date (employee_id, date) );

Tabla: audit_log

CREATE TABLE audit_log ( id VARCHAR(50) PRIMARY KEY, employee_id VARCHAR(50) NOT NULL, entity_type VARCHAR(50) NOT NULL, entity_id VARCHAR(50), action VARCHAR(50) NOT NULL, field_changed VARCHAR(100), old_value TEXT, new_value TEXT, changed_by VARCHAR(50) NOT NULL, changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (employee_id) REFERENCES employees(id), INDEX idx_employee_date (employee_id, changed_at) );
